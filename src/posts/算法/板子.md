# 技巧类
## 取消同步流
```cpp
ios::sync_with_stdio(false);
cin.tie(nullptr);
```

- **用途**：解除 C/C++ 流同步、解除 cin 与 cout 绑定。
- **性质**：cin/cout 提速约 3~5 倍，大输入必加；加了之后不要混用 scanf/printf。
```cpp
// 多组输入到 EOF
while (cin >> n >> m) { }

// 读整行（cin >> 后残留换行要先吞掉）
cin.ignore();
getline(cin, line);

printf("%.2f\n", x);   // 保留两位小数
```
##常见坑速查

| 坑 | 正确写法 |
|---|---|
| `1 << 62` 溢出 | `1LL << 62` |
| `accumulate` 初值 int 溢出 | 初值写 `0LL` |
| `mp[key]` 插入默认值 | 查询用 `find` / `count` |
| multiset `erase(x)` 全删 | `erase(find(x))` 删一个 |
| `v.size() - 1` 无符号下溢 | 先判空或 `(int)v.size() - 1` |
| 比较器写 `<=` 导致 RE | 严格弱序，用 `<` 和 `>` |
| 浮点判等 `==` | `fabs(a - b) < 1e-9` |
| cin 和 getline 混用 | 中间加 `cin.ignore()` |
| 树状数组下标 0 死循环 | 下标从 1 开始 |
| 线段树空间开 n | 开 `4n` |

## 中国剩余定理

适用处  同余方程组求解、大数重构。

```cpp
#include <vector>
using namespace std;

// Helper function for modular exponentiation (Fast Power)
long long quickPow(long long base, long long exp, long long mod) {
    long long res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = (res * base) % mod;
        base = (base * base) % mod;
        exp /= 2;
    }
    return res;
}

long long chineseRemainder(const vector<long long>& remainders, const vector<long long>& moduli) {
    long long prod = 1;
    for (auto m : moduli) prod *= m;

    long long result = 0;
    for (int i = 0; i < remainders.size(); i++) {
        long long pp = prod / moduli[i];

        // Fixed syntax: changed moduli[: to moduli[i]);
        // Calculates the modular inverse using Fermat's Little Theorem
        long long inv = quickPow(pp, moduli[i] - 2, moduli[i]);

        // Better approach: Apply modulo during multiplication to prevent overflow
        result = (result + remainders[i] * pp % prod * inv % prod) % prod;
    }

    return result;
}
```

## 线性筛（素数）

```cpp
const int N = 1e7 + 5;
int primes[N], cnt; bool vis[N];
void sieve(int n) {
    for (int i = 2; i <= n; i++) {
        if (!vis[i]) primes[cnt++] = i;
        for (int j = 0; primes[j] * i <= n; j++) {
            vis[primes[j] * i] = 1;
            if (i % primes[j] == 0) break;
        }
    }
}
```

```cpp
    for (int i = 1; i <= n; i++) fact[i] = fact[i-1] * i % MOD;
    ifact[n] = inv(fact[n]);
    for (int i = n; i >= 1; i--) ifact[i-1] = ifact[i] * i % MOD;
}
ll C(ll n, ll m) { return m < 0 || m > n ? 0 : fact[n] * ifact[m] % MOD * ifact[n-m] % MOD; }
```

- **用途**：计数 DP、二项式定理、放球模型。
- **性质**：预处理 O(n)，单次查询 O(1)；要求 MOD 为质数且 n < MOD（否则用 Lucas）

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const long long INF = 2e18;

struct Node {
    long long sum, lmax, rmax, ans;
    long long set_tag;
    int len;
};

int n, Q;
vector<Node> tree;
vector<int> arr;

// 向上合并信息
void pushup(int idx) {
    int lc = idx << 1, rc = idx << 1 | 1;
    tree[idx].sum = tree[lc].sum + tree[rc].sum;
    tree[idx].lmax = max(tree[lc].lmax, tree[lc].sum + tree[rc].lmax);
    tree[idx].rmax = max(tree[rc].rmax, tree[rc].sum + tree[lc].rmax);
    tree[idx].ans = max({tree[lc].ans, tree[rc].ans, tree[lc].rmax + tree[rc].lmax});
}

// 应用区间赋值操作
void apply_set(int idx, long long v) {
    tree[idx].set_tag = v;
    tree[idx].sum = (long long)tree[idx].len * v;
    if (v >= 0) {
        tree[idx].lmax = tree[idx].rmax = tree[idx].ans = tree[idx].sum;
    } else {
        tree[idx].lmax = tree[idx].rmax = tree[idx].ans = v;
    }
}

// 下传懒标记
void pushdown(int idx) {
    if (tree[idx].set_tag != INF) {
        apply_set(idx << 1, tree[idx].set_tag);
        apply_set(idx << 1 | 1, tree[idx].set_tag);
        tree[idx].set_tag = INF;
    }
}

// 建树
void build(int idx, int l, int r) {
    tree[idx].set_tag = INF;
    tree[idx].len = r - l + 1;
    if (l == r) {
        long long v = arr[l];
        tree[idx].sum = v;
        if (v >= 0) {
            tree[idx].lmax = tree[idx].rmax = tree[idx].ans = v;
        } else {
            tree[idx].lmax = tree[idx].rmax = tree[idx].ans = v;
        }
        return;
    }
    int mid = l + (r - l) / 2;
    build(idx << 1, l, mid);
    build(idx << 1 | 1, mid + 1, r);
    pushup(idx);
}

// 单点修改
void update_pos(int idx, int l, int r, int pos, long long v) {
    if (l == r) {
        tree[idx].sum = v;
        if (v >= 0) {
            tree[idx].lmax = tree[idx].rmax = tree[idx].ans = v;
        } else {
            tree[idx].lmax = tree[idx].rmax = tree[idx].ans = v;
        }
        return;
    }
    pushdown(idx);
    int mid = l + (r - l) / 2;
    if (pos <= mid) update_pos(idx << 1, l, mid, pos, v);
    else update_pos(idx << 1 | 1, mid + 1, r, pos, v);
    pushup(idx);
}

// 区间赋值
void update_range(int idx, int l, int r, int ql, int qr, long long v) {
    if (ql <= l && r <= qr) {
        apply_set(idx, v);
        return;
    }
    pushdown(idx);
    int mid = l + (r - l) / 2;
    if (ql <= mid) update_range(idx << 1, l, mid, ql, qr, v);
    if (qr > mid) update_range(idx << 1 | 1, mid + 1, r, ql, qr, v);
    pushup(idx);
}

// 区间查询
Node query(int idx, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) {
        return tree[idx];
    }
    pushdown(idx);
    int mid = l + (r - l) / 2;
    if (qr <= mid) return query(idx << 1, l, mid, ql, qr);
    if (ql > mid) return query(idx << 1 | 1, mid + 1, r, ql, qr);
    
    Node left = query(idx << 1, l, mid, ql, qr);
    Node right = query(idx << 1 | 1, mid + 1, r, ql, qr);
    Node res;
    res.sum = left.sum + right.sum;
    res.lmax = max(left.lmax, left.sum + right.lmax);
    res.rmax = max(right.rmax, right.sum + left.rmax);
    res.ans = max({left.ans, right.ans, left.rmax + right.lmax});
    res.len = 0; 
    return res;
}

int main() {
    // 优化输入输出流
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    if (!(cin >> n >> Q)) return 0;
    
    arr.resize(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> arr[i];
    }

    tree.resize(4 * n + 5);
    build(1, 1, n);

    while (Q--) {
        int op;
        cin >> op;
        if (op == 1) {
            int i;
            long long v;
            cin >> i >> v;
            update_pos(1, 1, n, i, v);
        } else if (op == 2) {
            int l, r;
            long long v;
            cin >> l >> r >> v;
            update_range(1, 1, n, l, r, v);
        } else {
            int l, r;
            cin >> l >> r;
            cout << query(1, 1, n, l, r).ans << "\n";
        }
    }

    return 0;
}
```

### 1.2 完整代码（DFS + BFS）

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 105;
const int INF = 0x3f3f3f3f;

int n, m;                       // n: 顶点数, m: 边数
int g[MAXN][MAXN];              // 邻接矩阵
bool vis[MAXN];                 // 访问标记

// 初始化图
void init() {
    memset(g, 0x3f, sizeof(g)); // 初始化为无穷大
    for (int i = 1; i <= n; i++) g[i][i] = 0;
}

// 添加无向边
void addEdge(int u, int v, int w) {
    g[u][v] = w;
    g[v][u] = w;
}

// 添加有向边
void addDirectedEdge(int u, int v, int w) {
    g[u][v] = w;
}

// DFS - 递归实现
void dfs_matrix(int u) {
    vis[u] = true;
    cout << u << " ";
    for (int v = 1; v <= n; v++) {
        if (g[u][v] != INF && g[u][v] != 0 && !vis[v]) {
            dfs_matrix(v);
        }
    }
}

// BFS - 队列实现
void bfs_matrix(int start) {
    memset(vis, false, sizeof(vis));
    queue<int> q;
    q.push(start);
    vis[start] = true;
    
    while (!q.empty()) {
        int u = q.front(); q.pop();
        cout << u << " ";
        for (int v = 1; v <= n; v++) {
            if (g[u][v] != INF && g[u][v] != 0 && !vis[v]) {
                vis[v] = true;
                q.push(v);
            }
        }
    }
}

// 测试
int main() {
    n = 5; m = 4;
    init();
    // 构建图: 1-2, 1-3, 2-4, 3-5
    addEdge(1, 2, 1);
    addEdge(1, 3, 1);
    addEdge(2, 4, 1);
    addEdge(3, 5, 1);
    
    cout << "邻接矩阵 DFS: ";
    memset(vis, false, sizeof(vis));
    dfs_matrix(1);
    cout << "\n邻接矩阵 BFS: ";
    bfs_matrix(1);
    return 0;
}
```
## 邻接表
```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;

int n, m;
vector<pair<int, int>> adj[MAXN];   // adj[u] = {(v, w), ...}
bool vis[MAXN];

// 添加有向边
void addDirectedEdge(int u, int v, int w) {
    adj[u].push_back({v, w});
}

// 添加无向边
void addEdge(int u, int v, int w) {
    adj[u].push_back({v, w});
    adj[v].push_back({u, w});
}

// DFS - 递归实现
void dfs_list(int u) {
    vis[u] = true;
    cout << u << " ";
    for (auto [v, w] : adj[u]) {
        if (!vis[v]) {
            dfs_list(v);
        }
    }
}

// DFS - 栈实现（非递归）
void dfs_list_stack(int start) {
    memset(vis, false, sizeof(vis));
    stack<int> stk;
    stk.push(start);
    
    while (!stk.empty()) {
        int u = stk.top(); stk.pop();
        if (vis[u]) continue;
        vis[u] = true;
        cout << u << " ";
        // 逆序入栈保证遍历顺序与递归一致
        for (int i = adj[u].size() - 1; i >= 0; i--) {
            int v = adj[u][i].first;
            if (!vis[v]) stk.push(v);
        }
    }
}

// BFS - 队列实现
void bfs_list(int start) {
    memset(vis, false, sizeof(vis));
    queue<int> q;
    q.push(start);
    vis[start] = true;
    
    while (!q.empty()) {
        int u = q.front(); q.pop();
        cout << u << " ";
        for (auto [v, w] : adj[u]) {
            if (!vis[v]) {
                vis[v] = true;
                q.push(v);
            }
        }
    }
}

// 测试
int main() {
    n = 5; m = 4;
    addEdge(1, 2, 1);
    addEdge(1, 3, 1);
    addEdge(2, 4, 1);
    addEdge(3, 5, 1);
    
    cout << "邻接表 DFS(递归): ";
    memset(vis, false, sizeof(vis));
    dfs_list(1);
    
    cout << "\n邻接表 DFS(栈):   ";
    dfs_list_stack(1);
    
    cout << "\n邻接表 BFS:      ";
    bfs_list(1);
    return 0;
}
```

## 邻接表的迪杰斯特拉
```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<ll, int> pli;          // <距离, 顶点>
const ll INF = 1e18;

int n, m, s;                        // s: 源点
vector<pair<int, int>> adj[100005]; // (邻接点, 边权)
ll dist[100005];                    // 最短路距离
bool vis[100005];                   // 是否已确定最短路

// Dijkstra 算法
void dijkstra(int start) {
    // 初始化
    for (int i = 1; i <= n; i++) {
        dist[i] = INF;
        vis[i] = false;
    }
    dist[start] = 0;
    
    // 小根堆: 按距离排序
    priority_queue<pli, vector<pli>, greater<pli>> pq;
    pq.push({0, start});
    
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        
        if (vis[u]) continue;       // 已处理过，跳过
        vis[u] = true;
        
        // 松弛操作
        for (auto [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
}

// 输出路径（可选）: 记录前驱节点
int pre[100005];
void dijkstra_with_path(int start) {
    for (int i = 1; i <= n; i++) {
        dist[i] = INF;
        vis[i] = false;
        pre[i] = -1;
    }
    dist[start] = 0;
    
    priority_queue<pli, vector<pli>, greater<pli>> pq;
    pq.push({0, start});
    
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        
        for (auto [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pre[v] = u;          // 记录前驱
                pq.push({dist[v], v});
            }
        }
    }
}

```

## Dijkstra（非负边权）

```cpp
vector<ll> dijkstra(int n, int s, vector<vector<pii>> &g) {
    vector<ll> dist(n + 1, LINF);
    priority_queue<pll, vector<pll>, greater<>> pq;
    dist[s] = 0, pq.push({0, s});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;                       // 懒惰删除
        for (auto [v, w] : g[u])
            if (dist[u] + w < dist[v])
                dist[v] = dist[u] + w, pq.push({dist[v], v});
    }
    return dist;
}
```

- **用途**：单源最短路（正权图）。
- **性质**：O(m log m)；**边权不能有负**，有负边用 SPFA。

## SPFA（负边权 / 判负环）

```cpp
bool spfa(int n, int s, vector<array<ll,3>> &edges, vector<ll> &dist) {
    vector<vector<pll>> g(n + 1);
    for (auto [u, v, w] : edges) g[u].push_back({v, w});
    dist.assign(n + 1, LINF);
    vector<int> cnt(n + 1); vector<bool> inq(n + 1);
    queue<int> q;
    dist[s] = 0, q.push(s), inq[s] = 1;
    while (!q.empty()) {
        int u = q.front(); q.pop(), inq[u] = 0;
        for (auto [v, w] : g[u])
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                if (!inq[v]) { q.push(v), inq[v] = 1; if (++cnt[v] > n) return true; }
            }
    }
    return false;   // true = 存在负环
}
```

- **用途**：含负权的最短路、差分约束系统、判负环。
- **性质**：最坏 O(nm)（可被卡），随机图实际很快；判负环标准方法：**某点入队次数 > n**。

## LCA（倍增）

```cpp
struct LCA {
    int n, LOG;
    vector<vector<int>> up; vector<int> dep;
    LCA(int n, vector<vector<int>> &g, int root = 1) : n(n), dep(n + 1) {
        LOG = __lg(n) + 1;
        up.assign(n + 1, vector<int>(LOG));
        dep[root] = 1;
        auto dfs = [&](auto self, int u, int f) -> void {
            up[u][0] = f;
            for (int i = 1; i < LOG; i++) up[u][i] = up[up[u][i-1]][i-1];
            for (int v : g[u]) if (v != f) dep[v] = dep[u] + 1, self(self, v, u);
        };
        dfs(dfs, root, root);
    }
    int lca(int u, int v) {
        if (dep[u] < dep[v]) swap(u, v);
        int d = dep[u] - dep[v];
        for (int i = LOG-1; i >= 0; i--) if (d >> i & 1) u = up[u][i];
        if (u == v) return u;
        for (int i = LOG-1; i >= 0; i--)
            if (up[u][i] != up[v][i]) u = up[u][i], v = up[v][i];
        return up[u][0];
    }
};
```

- **用途**：树上两点最近公共祖先（树上差分、树上路径问题的基础）。
- **性质**：预处理 O(n log n)，单次查询 O(log n)；链状深树注意递归栈深。

## 拓扑排序（Kahn）

```cpp
vector<int> topo(int n, vector<vector<int>> &g, vector<int> in) {
    queue<int> q; vector<int> order;
    for (int i = 1; i <= n; i++) if (!in[i]) q.push(i);
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : g[u]) if (--in[v] == 0) q.push(v);
    }
    return order;   // order.size() < n 说明有环
}
```

- **用途**：DAG 上 DP 前的处理、任务依赖、判环。
- **性质**：O(n + m)；只能用于有向图；结果个数 < n 即有环。

# 字符串

## KMP（前缀函数）

```cpp
vector<int> kmp(const string &s) {
    int n = s.size(); vector<int> f(n);
    for (int i = 1; i < n; i++) {
        int j = f[i-1];
        while (j && s[i] != s[j]) j = f[j-1];
        if (s[i] == s[j]) j++;
        f[i] = j;
    }
    return f;
}
```

- **用途**：模式匹配、最短循环节（`n - f[n-1]` 整除 n 时）、字符串周期。
- **性质**：O(n + m)；`f` 即最长相同前后缀长度。

## 字符串哈希

```cpp
struct Hash {
    unsigned long long base = 131, mod = (1ULL << 61) - 1;
    vector<unsigned long long> h, pw;
    Hash(const string &s) {
        int n = s.size();
        h.resize(n + 1), pw.resize(n + 1, 1);
        for (int i = 0; i < n; i++)
            pw[i+1] = pw[i] * base % mod,
            h[i+1] = (h[i] * base + s[i]) % mod;
    }
    unsigned long long get(int l, int r) {   // [l, r] 0-indexed
        return (h[r+1] - h[l] * pw[r-l+1] % mod + mod) % mod;
    }
};
```

- **用途**：O(1) 判断任意子串相等（二分答案 + 哈希求最长公共子串等）。
- **性质**：预处理 O(n)、查询 O(1)；有极小碰撞概率，关键题用双哈希。

## 背包

```cpp
// 01 背包：每件选或不选（逆序枚举容量）
vector<ll> dp(m + 1, 0);
for (int i = 1; i <= n; i++)
    for (int j = m; j >= v[i]; j--) dp[j] = max(dp[j], dp[j - v[i]] + w[i]);

// 完全背包：每件无限选（正序枚举容量）
for (int i = 1; i <= n; i++)
    for (int j = v[i]; j <= m; j++) dp[j] = max(dp[j], dp[j - v[i]] + w[i]);
```

- **用途**：01 背包——选/不选类最值与计数；完全背包——无限取物、凑硬币最少枚数。
- **性质**：均 O(nm)；枚举方向是关键（逆序=每件一次，正序=无限次）。

## LIS（最长上升子序列）

```cpp
vector<int> a(n), f;
for (int x : a) {
```

## sort / stable_sort —— 排序

```cpp
vector<int> v = {3, 1, 4, 1, 5};
sort(v.begin(), v.end());                  // {1, 1, 3, 4, 5}
sort(v.begin(), v.end(), greater<int>());  // {5, 4, 3, 1, 1} 降序
```

- **用途**：绝大多数排序场景；排序后才能用二分、去重、贪心扫描。

- **坑**：对结构体排序要么写比较器，要么给结构体重载 `<`。

## unique —— 相邻去重


- **用途**：数组去重；**离散化**的核心步骤（把大坐标映射到 0~m-1）。

## lower_bound / upper_bound —— 二分查找（最重要）

```cpp
vector<int> v = {1, 2, 2, 2, 5};   // 必须有序！

int p = lower_bound(v.begin(), v.end(), 2) - v.begin();  // 1，第一个 >= 2 的位置
int q = upper_bound(v.begin(), v.end(), 2) - v.begin();  // 4，第一个 > 2 的位置
int cnt = q - p;                                          // 2 的个数 = 3

// 判存在
bool ok = binary_search(v.begin(), v.end(), 2);           // true

// 没有时：lower_bound 返回第一个更大元素的位置（可当插入位置）
// 全部小于 x 时返回 v.end()，注意判断
```

- **用途**：二分答案的边界查找、离散化定位、有序表中查前驱后继、统计某值出现次数。
- **性质**：O(log n)，只对**有序**序列有效。
- **坑**：忘记排序结果是未定义的；返回值是迭代器，减 `v.begin()` 才是下标。

## nth_element —— 第 k 小（不排序）

```cpp
vector<int> v = {5, 2, 8, 1, 9, 3};
nth_element(v.begin(), v.begin() + 2, v.end());
// 执行后：v[2] 是排序后应处在位置 2 的元素（第 3 小 = 3）
// 且 v[0..1] 都 <= v[2]，v[3..] 都 >= v[2]，但左右内部无序

// 求中位数
nth_element(v.begin(), v.begin() + n / 2, v.end());
int median = v[n / 2];
```

- **用途**：求第 k 小/大、中位数、TopK（前 k 小），不需要完整排序时。
- **性质**：平均 O(n)（比 sort 的 O(n log n) 快）；不是稳定操作。

## next_permutation —— 枚举排列

```cpp
vector<int> v = {1, 2, 3};
sort(v.begin(), v.end());              // 必须先排成最小，才能枚举全部
do {
    // 处理当前排列 v：123 -> 132 -> 213 -> 231 -> 312 -> 321
} while (next_permutation(v.begin(), v.end()));
// 枚举完全排列后返回 false
```

- **用途**：n 很小（≤8）时暴力枚举全排列；求字典序下一个排列。
- **性质**：均摊 O(n)；初始序列必须是**最小的排列**才能枚举完所有情况。

## iota —— 填充递增序列

```cpp
vector<int> v(5);
iota(v.begin(), v.end(), 0);           // {0, 1, 2, 3, 4}
iota(v.begin(), v.end(), 1);           // {1, 2, 3, 4, 5}
```

- **用途**：并查集初始化 `fa`、给元素编号、构造初始排列。
- **性质**：O(n)，头文件 `<numeric>`。

## merge —— 归并

```cpp
vector<int> a = {1, 3, 5}, b = {2, 4, 6};
vector<int> c;
merge(a.begin(), a.end(), b.begin(), b.end(), back_inserter(c));
// c = {1, 2, 3, 4, 5, 6}
```

- **用途**：合并两个有序序列（归并排序的一步）、双指针合并。
- **性质**：O(n + m)；要求两个输入序列都有序。

## 自定义比较与排序

## 基本形式

```cpp
// 三种等价的降序写法
sort(v.begin(), v.end(), greater<int>());                       // 内置仿函数
sort(v.begin(), v.end(), [](int a, int b){ return a > b; });    // lambda（推荐）
```

- **含义**：比较器 `cmp(a, b)` 返回 `true` 表示 **a 应排在 b 前面**。
- **坑**：返回 `a >= b`（带等号）不满足严格弱序，相等元素时行为未定义，可能 RE——**比较器永远只写 `<` 和 `>`**。

## 双关键字排序

```cpp
// pair 数组：first 升序，first 相同时 second 降序
vector<pair<int,int>> v = {{1, 4}, {1, 2}, {3, 1}};
sort(v.begin(), v.end(), [](const auto &a, const auto &b){
    if (a.first != b.first) return a.first < b.first;
    return a.second > b.second;
});
// 结果：{1, 4}, {1, 2}, {3, 1}
```

```cpp
// 结构体排序：按分数降序，分数相同按编号升序
struct Student { int score, id; };
vector<Student> a;
sort(a.begin(), a.end(), [](const Student &x, const Student &y){
    if (x.score != y.score) return x.score > y.score;
    return x.id < y.id;
});
```

- **用途**：排行榜/排名题、区间调度（按右端点贪心）、按权值+下标去重。

## 区间贪心排序（高频）

```cpp
// 区间按右端点升序 —— 经典"最多不重叠区间"贪心
vector<pii> segs(n);
sort(segs.begin(), segs.end(), [](const pii &a, const pii &b){
    return a.second < b.second;
});

// 区间按左端点升序 —— 合并区间、扫描线
sort(segs.begin(), segs.end());   // pair 默认先比 first，直接 sort 即可
```

- **用途**：会议室/活动安排（右端点）、线段合并（左端点）、最小化最大等待（短作业优先）。

## 结构体重载 `<`（用于 set / map / priority_queue）

```cpp
struct Node {
    int w, id;
    // priority_queue 默认取"最大"，反着写 = 实现小根堆
    bool operator<(const Node &o) const { return w > o.w; }
};
priority_queue<Node> pq;                  // 现在是小根堆
set<Node> st;                             // 按此顺序有序

// 或者不用重载，priority_queue 直接传比较器：
struct cmp { bool operator()(const Node &a, const Node &b) const { return a.w > b.w; } };
priority_queue<Node, vector<Node>, cmp> pq2;
```

- **用途**：把自定义类型放进有序容器/堆；堆优化 Dijkstra 存 (距离, 点)。
- **性质**：set/priority_queue 的比较器语义是"谁在前"，**和 sort 相同**；priority_queue 是反直觉的——比较器说 `a` 小它就当大根堆顶，所以想小根堆就把"小"定义成权值大的方向（即写 `a.w > b.w`）。

## 稳定排序

```cpp
// 多人按年龄排序，同龄人保持输入时的先后顺序
stable_sort(a.begin(), a.end(), [](const Person &x, const Person &y){
    return x.age < y.age;
});
```

- **用途**：要求"相等元素保持原相对顺序"的题（多级排序的替代方案：先按次要键排，再 stable_sort 按主要键）。
- **性质**：O(n log n)，稳定；多次 stable_sort 叠加可实现多关键字排序。

## 对下标排序（防越界 / 保留原位置）

```cpp
// 对"下标数组"排序（不移动原数据，保留原始位置）：
vector<int> idx(n);
iota(idx.begin(), idx.end(), 0);
sort(idx.begin(), idx.end(), [&](int i, int j){ return a[i] < a[j]; });
// 之后 a[idx[0]], a[idx[1]], ... 就是排序结果，idx[i] 是第 i 小元素的原下标
```

- **用途**：既要排序结果又要原始位置（求逆序对、按权值排序后输出原编号等）。
- **坑**：比较器里访问外部数组时用 `[&]` 捕获，且保证 a/b 是合法下标。
